# 打包步骤

# 1. 打包镜像文件命令（本地）
docker build -t ewa:0.0.5 .

# 2. 打包镜像（公有）
docker tag ewa:0.0.5 ghuiqiao711/ewa:0.0.5

# 3. 推送到 Docker Hub (docker.com)
# 先登录 Docker Hub
docker login
# 或使用用户名密码登录
# docker login -u ghuiqiao711

# 推送镜像
docker push ghuiqiao711/ewa:0.0.5

# 如果需要推送到 latest 标签
docker tag ghuiqiao711/ewa:0.0.5 ghuiqiao711/ewa:latest
docker push ghuiqiao711/ewa:latest
# 查看日志
docker-compose logs -f web

# 重启服务
docker-compose restart

# 停止服务
docker-compose down

# 更新应用
git pull
docker-compose build
docker-compose up -d
docker-compose exec web python manage.py migrate
docker-compose exec web python manage.py collectstatic --noinput

# 强制部署方案
sudo docker compose up -d --build --force-recreate --remove-orphans

重新构建 Docker 镜像命令：
docker-compose down
docker-compose build --no-cache
docker-compose up -d

# 获取SSL证书
docker-compose run --rm certbot certonly --webroot \
  --webroot-path=/var/www/certbot \
  -d your_domain.com \
  --email your_email@example.com \
  --agree-tos

# 在nginx.conf中启用HTTPS配置（取消注释）
# 然后重启Nginx
docker-compose restart nginx


# 1. 安装Docker
curl -fsSL https://get.docker.com | sh
sudo usermod -aG docker $USER
newgrp docker

# 2. 上传项目代码到服务器
# 使用git clone或scp上传

# 3. 进入项目目录
cd EbbinghausAnywhere

# 4. 修改.env文件（重要！）
nano .env
# 设置DEBUG=False
# 设置ALLOWED_HOSTS=your_domain.com,localhost
# 修改SECRET_KEY

# 5. 构建并启动
docker-compose build
docker-compose up -d

# 6. 创建管理员账户
docker-compose exec web python manage.py createsuperuser

# 7. 查看服务状态
docker-compose ps
docker-compose logs -f web
